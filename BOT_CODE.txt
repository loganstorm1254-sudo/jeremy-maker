void touchLedWatchdog() {
  ledHoldUntil = millis() + LED_AUTO_OFF_MS;
}


void setupLedPwm() {
  // Higher freq = smoother; full duty = max brightness
  ledcSetup(LED_CH_BLUE, 20000, LED_PWM_BITS);
  ledcSetup(LED_CH_RED, 20000, LED_PWM_BITS);
  ledcSetup(LED_CH_YELLOW, 20000, LED_PWM_BITS);
  ledcAttachPin(blueLED, LED_CH_BLUE);
  ledcAttachPin(redLED, LED_CH_RED);
  ledcAttachPin(yellowLED, LED_CH_YELLOW);
  ledcWrite(LED_CH_BLUE, 0);
  ledcWrite(LED_CH_RED, 0);
  ledcWrite(LED_CH_YELLOW, 0);
  pinMode(blueLEDK, OUTPUT);
  pinMode(redLEDK, OUTPUT);
  pinMode(yellowLEDK, OUTPUT);
  digitalWrite(blueLEDK, LOW);
  digitalWrite(redLEDK, LOW);
  digitalWrite(yellowLEDK, LOW);
}

void setLedPair(int pwmCh, int cathode, bool on) {
  pinMode(cathode, OUTPUT);
  digitalWrite(cathode, LOW); // always sink when using 2-pin LED drive
  if (on) {
    ledcWrite(pwmCh, LED_PWM_ON);
    touchLedWatchdog();
  } else {
    ledcWrite(pwmCh, 0);
  }
}

void setBlue(bool on) { setLedPair(LED_CH_BLUE, blueLEDK, on); }
void setRed(bool on) { setLedPair(LED_CH_RED, redLEDK, on); }
void setYellow(bool on) { setLedPair(LED_CH_YELLOW, yellowLEDK, on); }

void setAllLeds(bool on) {
  ledOverride = redOverride = yellowOverride = true;
  ledOverrideOn = redOverrideOn = yellowOverrideOn = on;
  setBlue(on);
  setRed(on);
  setYellow(on);
}

void applySwitchLeds() {
  ledOverride = redOverride = yellowOverride = true;
  ledOverrideOn = switchLedBlue;
  redOverrideOn = switchLedRed;
  yellowOverrideOn = switchLedYellow;
  setBlue(switchLedBlue);
  setRed(switchLedRed);
  setYellow(switchLedYellow);
}

// Blocks until switch opens: programmed screen text + selected LEDs
void holdSwitchProgramUntilOff() {
  applySwitchLeds();
  showSwitchScreen();
  while (digitalRead(switchPin) == LOW) {
    serviceUsbProgrammer();
    applySwitchLeds();
    showSwitchScreen();
    delay(40);
  }
  setAllLeds(false);
  clearLedOverrides();
}

void blazeAllLeds(bool on) {
  // Effects: ALWAYS hit blue + red + yellow together
  setAllLeds(on);
}

void enforceLedSafety(unsigned long now) {
  digitalWrite(boomPin, LOW); // never leave boom trigger stuck
  if (ledHoldUntil == 0) return;
  if (now < ledHoldUntil) return;
  if (ledOverride || redOverride || yellowOverride) {
    ledOverride = redOverride = yellowOverride = false;
    ledOverrideOn = redOverrideOn = yellowOverrideOn = false;
    setBlue(false);
    setRed(false);
    setYellow(false);
  }
  ledHoldUntil = 0;
}

// Effect strobe: all three LEDs mega-bright together (not blue-only chase)
void pulseAllLeds(unsigned long elapsed, unsigned long stepMs) {
  bool on = ((elapsed / stepMs) % 2) == 0;
  blazeAllLeds(on);
}

void clearLedOverrides() {
  ledOverride = redOverride = yellowOverride = false;
  ledOverrideOn = redOverrideOn = yellowOverrideOn = false;
  setBlue(false);
  setRed(false);
  setYellow(false);
  ledHoldUntil = 0;
}

}

function fileToMonoB64(file){
  return new Promise((resolve,reject)=>{
    const img=new Image();
    const url=URL.createObjectURL(file);
    img.onload=()=>{
      try{
        const c=document.getElementById('pngPreview');
        const ctx=c.getContext('2d');
        ctx.fillStyle='#000';
        ctx.fillRect(0,0,128,64);
        const scale=Math.min(128/img.width,64/img.height);
        const w=img.width*scale,h=img.height*scale;
        ctx.drawImage(img,(128-w)/2,(64-h)/2,w,h);
        const data=ctx.getImageData(0,0,128,64).data;
        const bytes=new Uint8Array(1024);
        for(let y=0;y<64;y++){
          for(let x=0;x<128;x++){
            const i=(y*128+x)*4;
            const lum=0.299*data[i]+0.587*data[i+1]+0.114*data[i+2];
            if(lum>=128) bytes[y*16+(x>>3)] |= (0x80>>(x&7));
          }
        }
        let bin='';
        for(let i=0;i<bytes.length;i++) bin+=String.fromCharCode(bytes[i]);

void generateCode(char* out, size_t len) {
  static const char alphabet[] = "ABCDEFGHJKLMNPQRSTUVWXYZ23456789";
  for (size_t i = 0; i + 1 < len; i++) {
    out[i] = alphabet[random(0, (int)(sizeof(alphabet) - 1))];
  }
  out[len - 1] = '\0';
}

bool codesEqual(const String& a, const char* b) {
  String left = a;
  left.trim();
  left.toUpperCase();
  String right = String(b);
  right.trim();
  right.toUpperCase();
  return left.length() > 0 && left == right;
}

String readPostBody() {
  if (server.hasArg("plain") && server.arg("plain").length() > 0) {
    return server.arg("plain");
  }
  // Fallbacks when clients POST form fields instead of JSON
  if (server.hasArg("code")) {
    String out = "{\"code\":\"";
    out += server.arg("code");
    out += "\"";
    if (server.hasArg("action")) {
      out += ",\"action\":\"";
      out += server.arg("action");
      out += "\"";
    }
    if (server.hasArg("text")) {
      out += ",\"text\":\"";
      out += server.arg("text");
      out += "\"";
    }
    if (server.hasArg("image")) {
      out += ",\"image\":\"";
      out += server.arg("image");
      out += "\"";
    }
    if (server.hasArg("seconds")) {
      out += ",\"seconds\":";
      out += server.arg("seconds");
    }
    out += "}";
    return out;
  }
  return "";
}

void upperCopy(char* dest, size_t destLen, const String& src) {
  size_t n = src.length();
  if (n >= destLen) n = destLen - 1;
  for (size_t i = 0; i < n; i++) {
    char c = src[i];
    if (c >= 'a' && c <= 'z') c = c - 32;
    dest[i] = c;
  }
  dest[n] = '\0';
}

String jsonEscape(const String& s) {
  String out;
  for (size_t i = 0; i < s.length(); i++) {
    char c = s[i];
    if (c == '"' || c == '\\') out += '\\';
    out += c;
  }
  return out;
}

String extractJsonString(const String& body, const char* key) {
  String needle = String("\"") + key + "\"";
  int idx = body.indexOf(needle);
  if (idx < 0) return "";
  int colon = body.indexOf(':', idx);
  if (colon < 0) return "";
  int i = colon + 1;
  while (i < (int)body.length() && (body[i] == ' ' || body[i] == '\t')) i++;
  if (i >= (int)body.length()) return "";
  if (body[i] == '"') {
    int q1 = i;
    int q2 = body.indexOf('"', q1 + 1);
    if (q2 <= q1) return "";
    return body.substring(q1 + 1, q2);
  }
  // number / bare token until comma or brace
  int end = i;
  while (end < (int)body.length() && body[end] != ',' && body[end] != '}' && body[end] != ' ') end++;
  return body.substring(i, end);
}

int extractJsonInt(const String& body, const char* key, int fallback) {
  String s = extractJsonString(body, key);
  if (s.length() == 0) return fallback;
  return constrain(s.toInt(), 1, 120);
}

int b64Val(char c) {
  if (c >= 'A' && c <= 'Z') return c - 'A';
  if (c >= 'a' && c <= 'z') return c - 'a' + 26;
  if (c >= '0' && c <= '9') return c - '0' + 52;
  if (c == '+') return 62;
  if (c == '/') return 63;
  return -1;
}

int decodeBase64(const String& in, uint8_t* out, int outMax) {
  int len = in.length();
  int o = 0;
  int val = 0;
  int valb = -8;
  for (int i = 0; i < len; i++) {
    char c = in[i];
    if (c == '=' || c == '\n' || c == '\r' || c == ' ') continue;
    int d = b64Val(c);
    if (d < 0) continue;
    val = (val << 6) + d;
    valb += 6;
    if (valb >= 0) {
      if (o >= outMax) return -1;
      out[o++] = (uint8_t)((val >> valb) & 0xFF);
      valb -= 8;
    }
  }
  return o;
}

bool nameIs(const char* a, const char* b) {
  if (!a || !b) return false;
  while (*a && *b) {
    char ca = (*a >= 'A' && *a <= 'Z') ? (*a + 32) : *a;
    char cb = (*b >= 'A' && *b <= 'Z') ? (*b + 32) : *b;
    if (ca != cb) return false;
    a++;
    b++;
  }
  return *a == *b;
}

void drawIcon16(const uint8_t* icon, int ox, int oy, int scale) {
  for (int y = 0; y < 16; y++) {
    uint8_t hi = pgm_read_byte(&icon[y * 2]);
    uint8_t lo = pgm_read_byte(&icon[y * 2 + 1]);
    uint16_t row = ((uint16_t)hi << 8) | lo;
    for (int x = 0; x < 16; x++) {
      if (row & (0x8000 >> x)) {
        display.fillRect(ox + x * scale, oy + y * scale, scale, scale, SH110X_WHITE);
      }
    }
  }
}

void drawWrappedText(const char* text, int yStart) {
  display.setTextSize(1);
  display.setTextColor(SH110X_WHITE);
  String s = String(text);
  s.toUpperCase();
  int maxChars = 21;
  // SH1106 clips the top row — never draw above y=8
  int y = yStart < 8 ? 8 : yStart;
  while (s.length() > 0 && y < 56) {
    String line = s.substring(0, maxChars);
    int cut = line.lastIndexOf(' ');
    if (s.length() > (unsigned)maxChars && cut > 8) {
      line = s.substring(0, cut);
      s = s.substring(cut + 1);
    } else {
      s = s.substring(line.length());
      s.trim();
    }
    int16_t x1, y1;
    uint16_t w, h;
    display.getTextBounds(line.c_str(), 0, 0, &x1, &y1, &w, &h);
    display.setCursor((128 - (int)w) / 2, y);
    display.print(line);
    y += 12;
  }
}

void renderOverlay() {
  display.clearDisplay();

    for (int y = 0; y < 64; y++) {
      for (int x = 0; x < 128; x++) {
        uint8_t byte = overlayBitmap[y * 16 + (x >> 3)];
        if (byte & (0x80 >> (x & 7))) {
          display.drawPixel(x, y, SH110X_WHITE);
        }
      }

    drawIcon16(iconByName(overlayImage), 32, 10, 3);
    if (overlayText[0]) {
      display.setCursor(2, 56);
      display.print(overlayText);
    }
  } else {
    drawWrappedText(overlayText, 18);
  }
  display.display();
}

void handleAuth() {
  sendCors();
  if (server.method() != HTTP_POST) {
    server.send(405, "application/json", "{\"ok\":false,\"message\":\"POST ONLY\"}");
    return;
  }

  String body = readPostBody();
  String code = extractJsonString(body, "code");
  code.trim();
  code.toUpperCase();
  if (code.length() == 0) {
    server.send(400, "application/json", "{\"ok\":false,\"message\":\"MISSING CODE\"}");
    return;
  }

  // First-time pairing: OLED code becomes the panel session (persisted)
  if (pairingCodeOk(code)) {
    isPaired = true;
    needPairingScreen = false;
    prefs.putBool("paired", true);
    prefs.putBool("needPair", false);
    prefs.putInt("authGate", 1);
    prefs.remove("pairCode");
    lockPanelToPairingCode();
    if (state == PAIRING || state == INFO_SCREEN) state = NORMAL;
    String resp = "{\"ok\":true,\"message\":\"BOT AUTHENTICATED\",\"paired\":true,\"panel\":true,\"panelCode\":\"";
    resp += panelCode;
    resp += "\"}";
    server.send(200, "application/json", resp);
    return;
  }

  // Already paired: only the CURRENT panel code (never rewrite it from the client)
  if (panelCodeOk(code)) {
    String resp = "{\"ok\":true,\"message\":\"PANEL CODE ACCEPTED\",\"paired\":true,\"panel\":true,\"panelCode\":\"";
    resp += panelCode;
    resp += "\"}";
    server.send(200, "application/json", resp);
    return;
  }

  if (isPaired && !panelCodeActive) {
    server.send(401, "application/json",
                "{\"ok\":false,\"message\":\"DOUBLE-CLICK BOOT ON THE BOT FOR A PANEL CODE\"}");
    return;
  }

  server.send(401, "application/json",
              "{\"ok\":false,\"message\":\"WRONG CODE — CHECK THE ROBOT SCREEN\"}");
}

void handleCommand() {
  sendCors();
  if (server.method() != HTTP_POST) {
    server.send(405, "application/json", "{\"ok\":false,\"message\":\"POST ONLY\"}");
    return;
  }

  String body = readPostBody();
  String code = extractJsonString(body, "code");
  code.trim();
  code.toUpperCase();
  String action = extractJsonString(body, "action");
  action.toLowerCase();

  if (!isPaired) {
    server.send(401, "application/json",
                "{\"ok\":false,\"message\":\"AUTH BOT FIRST — PAIRING LOCKED\"}");
    return;
  }
  if (!codeAccepted(code)) {
    server.send(401, "application/json",
                "{\"ok\":false,\"message\":\"WRONG CODE — CHECK THE CODE ON THE ROBOT SCREEN\"}");
    return;
  }
  if (action == "led_on") {
    ledOverride = true; ledOverrideOn = true; setBlue(true);
    server.send(200, "application/json", "{\"ok\":true,\"message\":\"BLUE ON\"}");
    return;
  }
  if (action == "led_off") {
    ledOverride = true; ledOverrideOn = false; setBlue(false);
    server.send(200, "application/json", "{\"ok\":true,\"message\":\"BLUE OFF\"}");
    return;
  }
  if (action == "red_on") {
    redOverride = true; redOverrideOn = true; setRed(true);
    server.send(200, "application/json", "{\"ok\":true,\"message\":\"RED ON\"}");
    return;
  }
  if (action == "red_off") {
    redOverride = true; redOverrideOn = false; setRed(false);
    server.send(200, "application/json", "{\"ok\":true,\"message\":\"RED OFF\"}");
    return;
  }
  if (action == "yellow_on") {
    yellowOverride = true; yellowOverrideOn = true; setYellow(true);
    server.send(200, "application/json", "{\"ok\":true,\"message\":\"YELLOW ON\"}");
    return;
  }
  if (action == "yellow_off") {
    yellowOverride = true; yellowOverrideOn = false; setYellow(false);
    server.send(200, "application/json", "{\"ok\":true,\"message\":\"YELLOW OFF\"}");
    return;

    ledOverride = redOverride = yellowOverride = true;

    return;

                  "{\"ok\":false,\"message\":\"BOOM COOLING DOWN — WAIT\"}");
      return;

    lastBoomMs = nowBoom;

    setAllLeds(true);
    delay(BOOM_PULSE_MS);
    digitalWrite(boomPin, LOW);
    startAnim(ANIM_BOOM, 900);
    server.send(200, "application/json", "{\"ok\":true,\"message\":\"BOOM\"}");
    return;
  }
  if (action == "pop") {
    forcePop = true;
    server.send(200, "application/json", "{\"ok\":true,\"message\":\"EYES POPPING\"}");
    return;

    return;

    startAnim(ANIM_ORBIT_BLINK, 2800);
    server.send(200, "application/json", "{\"ok\":true,\"message\":\"ORBIT BLINK\"}");
    return;

  if (action == "midnight_burst") {

    startAnim(ANIM_MIDNIGHT_BURST, 4500);
    server.send(200, "application/json", "{\"ok\":true,\"message\":\"MIDNIGHT BURST\"}");
    return;

    startAnim(ANIM_SPARK_SALUTE, 3200);
    server.send(200, "application/json", "{\"ok\":true,\"message\":\"SPARK SALUTE\"}");
    return;
  }
  if (action == "confetti_pop") {
    startAnim(ANIM_CONFETTI_POP, 2800);
    server.send(200, "application/json", "{\"ok\":true,\"message\":\"CONFETTI POP\"}");
    return;
  }

    overlayUntil = 0;
    overlayIsBitmap = false;
    activeAnim = ANIM_NONE;
    state = resumeAfterTransient();
    if (state == SHOW_OVERLAY || state == PLAY_ANIM) state = isPaired ? NORMAL : PAIRING;
    server.send(200, "application/json", "{\"ok\":true,\"message\":\"SCREEN CLEARED\"}");
    return;
  }
  if (action == "show_text") {
    String text = extractJsonString(body, "text");
    text.trim();
    text.toUpperCase();
    if (text.length() == 0) {
      server.send(400, "application/json", "{\"ok\":false,\"message\":\"MISSING TEXT\"}");
      return;
    }
    if (text.length() > 40) text = text.substring(0, 40);
    int seconds = extractJsonInt(body, "seconds", 3);
    startOverlayText(text.c_str(), seconds);
    server.send(200, "application/json",
                "{\"ok\":true,\"message\":\"SHOWING TEXT ON OLED\"}");
    return;
  }
  if (action == "show_image") {
    String image = extractJsonString(body, "image");
    image.trim();
    image.toLowerCase();
    if (image.length() == 0) image = "star";
    int seconds = extractJsonInt(body, "seconds", 3);
    startOverlayImage(image.c_str(), seconds);
    server.send(200, "application/json",
                "{\"ok\":true,\"message\":\"SHOWING IMAGE ON OLED\"}");
    return;
  }

  if (action == "show_png" || action == "show_bitmap") {
    String b64 = extractJsonString(body, "bitmap");
    if (b64.length() == 0) b64 = extractJsonString(body, "png");
    b64.trim();
    if (b64.length() < 8) {
      server.send(400, "application/json", "{\"ok\":false,\"message\":\"MISSING BITMAP\"}");
      return;
    }
    memset(overlayBitmap, 0, OLED_BMP_BYTES);
    int n = decodeBase64(b64, overlayBitmap, OLED_BMP_BYTES);
    if (n < OLED_BMP_BYTES) {
      server.send(400, "application/json", "{\"ok\":false,\"message\":\"BAD BITMAP SIZE\"}");
      return;
    }
    int seconds = extractJsonInt(body, "seconds", 5);
    startOverlayBitmap(seconds);
    server.send(200, "application/json",
                "{\"ok\":true,\"message\":\"SHOWING PNG ON OLED\"}");
    return;
  }

  server.send(400, "application/json", "{\"ok\":false,\"message\":\"UNKNOWN ACTION\"}");
}

void drawEye(int x, int y, float s) {
  int half = (int)(s / 2.0f);
  if (half < 2) return;
  int w = (int)s;
  int h = (int)s;
  int ox2 = 0;
  int oy2 = 0;

  switch (eyeShape) {
    case 0: // circle
      display.fillCircle(x, y, half, SH110X_WHITE);
      break;
    case 1: // square
      display.fillRect(x - half, y - half, w, h, SH110X_WHITE);
      break;
    case 3: // wide
      ox2 = 4;
      oy2 = -2;
      display.fillRoundRect(x - half - ox2, y - half - oy2, w + ox2 * 2, h + oy2 * 2, 6, SH110X_WHITE);
      break;
    case 4: // tall
      ox2 = -2;
      oy2 = 3;
      display.fillRoundRect(x - half - ox2, y - half - oy2, w + ox2 * 2, h + oy2 * 2, 6, SH110X_WHITE);
      break;
    case 5: // diamond
      display.fillTriangle(x, y - half, x + half, y, x, y + half, SH110X_WHITE);
      display.fillTriangle(x, y - half, x - half, y, x, y + half, SH110X_WHITE);
      break;
    default: // soft square
      display.fillRoundRect(x - half, y - half, w, h, 10, SH110X_WHITE);
      break;
  }

  int ps = (int)pupilSize;
  if (ps < 1) ps = 1;
  if (ps > 12) ps = 12;
  switch (pupilStyle) {
    case 0:
      break;
    case 1:
      display.fillCircle(x, y, ps, SH110X_BLACK);
      break;
    case 2:
      display.drawCircle(x, y, ps, SH110X_BLACK);
      if (ps > 2) display.drawCircle(x, y, ps - 1, SH110X_BLACK);
      break;
    case 3:
      display.fillRect(x - ps, y - 1, ps * 2, 3, SH110X_BLACK);
      break;
    case 4:
      display.fillRect(x - ps, y - 1, ps * 2, 3, SH110X_BLACK);
      display.fillRect(x - 1, y - ps, 3, ps * 2, SH110X_BLACK);
      break;
    case 5:
      display.drawLine(x - ps, y, x + ps, y, SH110X_BLACK);
      display.drawLine(x, y - ps, x, y + ps, SH110X_BLACK);
      display.fillCircle(x, y, ps / 2 > 0 ? ps / 2 : 1, SH110X_BLACK);
      break;
    default:
      display.fillCircle(x, y, ps, SH110X_BLACK);
      break;
  }
}

void showSwitchScreen() {
  display.clearDisplay();
  display.setTextColor(SH110X_WHITE);
  display.setTextSize(1);

  // Wrap programmed text into up to 4 lines of 21 chars (128px / 6)
  const char* src = switchScreenText;
  if (!src[0]) src = "JEREMY CO";
  int line = 0;
  int col = 0;
  char row[22];
  row[0] = 0;
  for (int i = 0; src[i] && line < 4; i++) {
    char c = src[i];
    if (c == '\n' || col >= 21) {
      row[col] = 0;
      display.setCursor(0, 8 + line * 12);
      display.print(row);
      line++;
      col = 0;
      row[0] = 0;
      if (c == '\n') continue;
    }
    if (c >= 32 && c <= 126) {
      row[col++] = c;
      row[col] = 0;
    }
  }
  if (col > 0 && line < 4) {
    display.setCursor(0, 8 + line * 12);
    display.print(row);
  }
  display.display();
}

}

// Raw-line handler (preserves text case)
void handleUsbRawLine(const String& raw) {
  String line = raw;
  line.trim();
  if (line.length() == 0) return;

  String up = line;
  up.toUpperCase();

  if (up == "HELLO" || up == "PING") {
    Serial.print("JEREMY_OK|");
    Serial.println(FIRMWARE_VERSION);
    return;
  }
  if (up == "GETSWITCH") {
    Serial.print("SWITCH|");
    Serial.print(switchScreenText);
    Serial.print("|");
    Serial.print(switchLedBlue ? 1 : 0);
    Serial.print("|");
    Serial.print(switchLedRed ? 1 : 0);
    Serial.print("|");
    Serial.println(switchLedYellow ? 1 : 0);
    return;
  }
  if (up == "GETLOOK") {
    Serial.print("LOOK|");
    Serial.print(eyeShape);
    Serial.print("|");
    Serial.print(pupilStyle);
    Serial.print("|");
    Serial.print(pupilSize);
    Serial.print("|");
    Serial.print(cornerTL);
    Serial.print("|");
    Serial.print(cornerTR);
    Serial.print("|");
    Serial.print(cornerBL);
    Serial.print("|");
    Serial.print(cornerBR);
    Serial.print("|");
    Serial.println(lookShowUptime ? 1 : 0);
    return;
  }
  if (up.startsWith("SETLOOK|")) {
    // SETLOOK|shape|pupil|size|tl|tr|bl|br|uptime
    int p[9];
    p[0] = line.indexOf('|');
    for (int i = 1; i < 9; i++) {
      p[i] = (p[i - 1] >= 0) ? line.indexOf('|', p[i - 1] + 1) : -1;
    }
    // Need 8 pipes → indices 0..7 at minimum; uptime is after p[7]
    if (p[0] < 0 || p[1] < 0 || p[2] < 0 || p[3] < 0 || p[4] < 0 || p[5] < 0 || p[6] < 0 || p[7] < 0) {
      Serial.println("ERR|BAD_SETLOOK");
      return;
    }
    eyeShape = (uint8_t)constrain(line.substring(p[0] + 1, p[1]).toInt(), 0, 5);
    pupilStyle = (uint8_t)constrain(line.substring(p[1] + 1, p[2]).toInt(), 0, 5);
    pupilSize = (uint8_t)constrain(line.substring(p[2] + 1, p[3]).toInt(), 2, 10);
    copyCorner(cornerTL, sizeof(cornerTL), line.substring(p[3] + 1, p[4]));
    copyCorner(cornerTR, sizeof(cornerTR), line.substring(p[4] + 1, p[5]));
    copyCorner(cornerBL, sizeof(cornerBL), line.substring(p[5] + 1, p[6]));
    copyCorner(cornerBR, sizeof(cornerBR), line.substring(p[6] + 1, p[7]));
    lookShowUptime = true; // ignore host flag — uptime/version always on
    saveLookProgram();
    Serial.print("LOOKOK|");
    Serial.println(FIRMWARE_VERSION);
    return;
  }
  if (up.startsWith("SETSW|")) {
    // SETSW|B|R|Y|text  — text keeps original case from `line`
    int p1 = line.indexOf('|');
    int p2 = line.indexOf('|', p1 + 1);
    int p3 = line.indexOf('|', p2 + 1);
    int p4 = line.indexOf('|', p3 + 1);
    if (p1 < 0 || p2 < 0 || p3 < 0 || p4 < 0) {
      Serial.println("ERR|BAD_SETSW");
      return;
    }
    switchLedBlue = line.substring(p1 + 1, p2).toInt() != 0;
    switchLedRed = line.substring(p2 + 1, p3).toInt() != 0;
    switchLedYellow = line.substring(p3 + 1, p4).toInt() != 0;
    String text = line.substring(p4 + 1);
    sanitizeSwitchText(text);
    strncpy(switchScreenText, text.c_str(), sizeof(switchScreenText) - 1);
    switchScreenText[sizeof(switchScreenText) - 1] = 0;
    saveSwitchProgram();
    Serial.print("OK|");
    Serial.println(switchScreenText);
    return;
  }
  if (up == "SAVE") {
    saveSwitchProgram();
    Serial.println("SAVED");
    return;
  }
  Serial.println("ERR|UNKNOWN");
}

void serviceUsbProgrammer() {
  while (Serial.available() > 0) {
    char c = (char)Serial.read();
    if (c == '\n' || c == '\r') {
      if (usbLineBuf.length() > 0) {
        handleUsbRawLine(usbLineBuf);
        usbLineBuf = "";
      }
    } else if (usbLineBuf.length() < 160) {
      usbLineBuf += c;
    }
  }
}

// delay() blocks USB replies — Maker pings during boot after port.open() resets the ESP
void delayServicingUsb(unsigned long ms) {
  unsigned long start = millis();
  while (millis() - start < ms) {
    serviceUsbProgrammer();
    delay(20);
  }
}

void tickSparks(float gravity) {
  for (int i = 0; i < MAX_SPARKS; i++) {
    if (!sparks[i].alive) continue;
    sparks[i].x += sparks[i].vx;
    sparks[i].y += sparks[i].vy;
    sparks[i].vy += gravity;
    sparks[i].life--;
    if (sparks[i].life <= 0 || sparks[i].x < -2 || sparks[i].x > 130 ||
        sparks[i].y < -2 || sparks[i].y > 70) {
      sparks[i].alive = false;
    }
  }
}

void drawSparks() {
  for (int i = 0; i < MAX_SPARKS; i++) {
    if (!sparks[i].alive) continue;
    int r = (sparks[i].life > 20) ? 2 : 1;
    display.fillCircle((int)sparks[i].x, (int)sparks[i].y, r, SH110X_WHITE);
  }
}

bool anySparksAlive() {
  for (int i = 0; i < MAX_SPARKS; i++) if (sparks[i].alive) return true;
  return false;
}

void runAnimFrame() {
  unsigned long now = millis();
  unsigned long elapsed = now - animStart;

  if (now >= animUntil && activeAnim != ANIM_MIDNIGHT_BURST &&
      activeAnim != ANIM_SPARK_SALUTE && activeAnim != ANIM_CONFETTI_POP) {
    activeAnim = ANIM_NONE;
    clearLedOverrides();
    state = resumeAfterTransient();
    if (state == SHOW_OVERLAY || state == PLAY_ANIM) state = isPaired ? NORMAL : PAIRING;
    return;
  }

  display.clearDisplay();

  switch (activeAnim) {
    case ANIM_LED_STROBE: {
      bool on = ((elapsed / 80) % 2) == 0;
      ledOverride = redOverride = yellowOverride = true;
      ledOverrideOn = redOverrideOn = yellowOverrideOn = on;
      setBlue(on);
      setRed(on);
      setYellow(on);
      display.setTextSize(2);
      display.setTextColor(SH110X_WHITE);
      display.setCursor(18, 24);
      display.print(on ? "STROBE" : "      ");
      break;
    }
    case ANIM_MIDNIGHT_BURST: {
      // New Cycle fireworks — Vector-style salute, Jeremy branded
      pulseAllLeds(elapsed, 120);

      if (elapsed < 900) {
        // rocket climb
        int rocketY = 56 - (int)(elapsed / 18);
        display.fillRect(62, rocketY, 3, 8, SH110X_WHITE);
        display.drawPixel(63, rocketY - 2, SH110X_WHITE);
      } else if (elapsed < 1100 && !anySparksAlive()) {
        spawnBurst(64, 18, 18, 2.8f);
      } else if (elapsed > 1800 && elapsed < 2000) {
        spawnBurst(32, 22, 10, 2.2f);
        spawnBurst(96, 26, 10, 2.2f);
      } else if (elapsed > 2800 && elapsed < 3000) {
        spawnBurst(64, 30, 14, 2.0f);
      }

      tickSparks(0.06f);
      drawSparks();

      display.setTextSize(1);
      display.setTextColor(SH110X_WHITE);
      if (elapsed > 3200) {
        display.setCursor(16, 28);
        display.print("NEW CYCLE");
        display.setCursor(28, 42);
        display.print("JEREMY");
      }

      if (elapsed > 4500 && !anySparksAlive()) {
        activeAnim = ANIM_NONE;
        clearLedOverrides();
        state = resumeAfterTransient();
      }
      break;
    }
    case ANIM_SPARK_SALUTE: {
      pulseAllLeds(elapsed, 90);
      if ((elapsed % 400) < 40) spawnBurst(20 + random(0, 90), 20 + random(0, 20), 6, 1.8f);
      tickSparks(0.05f);
      drawSparks();
      drawIcon16(ICON_BURST, 48, 16, 2);
      if (elapsed > 3200) {
        activeAnim = ANIM_NONE;
        clearLedOverrides();
        state = resumeAfterTransient();
      }
      break;
    }
    case ANIM_ORBIT_BLINK: {
      float ang = elapsed / 180.0f;
      float r = 18;
      int cx = 64 + (int)(cosf(ang) * r);
      int cy = 32 + (int)(sinf(ang) * r * 0.55f);
      drawEye(cx - 18, 32, 18 + 4 * sinf(ang * 2));
      drawEye(cx + 18, 32, 18 + 4 * cosf(ang * 2));
      if (((elapsed / 200) % 5) == 0) {
        // blink lids
        display.fillRect(cx - 30, 28, 60, 8, SH110X_BLACK);
      }
      pulseAllLeds(elapsed, 80);
      break;
    }
    case ANIM_EYE_WIGGLE: {
      float w = sinf(elapsed / 70.0f) * 14;
      float h = cosf(elapsed / 90.0f) * 8;
      drawEye(leftX + (int)w, centerY + (int)h, BASE);
      drawEye(rightX - (int)w, centerY - (int)h, BASE);
      pulseAllLeds(elapsed, 70);
      break;
    }
    case ANIM_CONFETTI_POP: {
      if ((elapsed % 200) < 30) {
        for (int k = 0; k < 4; k++) {
          spawnBurst(random(10, 118), random(8, 50), 1, 1.2f);
        }
      }
      tickSparks(0.08f);
      drawSparks();
      display.setTextSize(1);
      display.setCursor(34, 28);
      display.print("POP!");
      setAllLeds(((elapsed / 90) % 2) == 0);
      if (elapsed > 2800) {
        activeAnim = ANIM_NONE;
        clearLedOverrides();
        state = resumeAfterTransient();
      }
      break;
    }
    case ANIM_BOOM: {
      // All three LEDs strobe hard — hardware boom pulse already fired
      setAllLeds(((elapsed / 50) % 2) == 0);
      digitalWrite(boomPin, LOW); // keep trigger off after pulse
      display.setTextSize(2);
      display.setTextColor(SH110X_WHITE);
      display.setCursor(28, 22);
      display.print("BOOM");
      if ((elapsed % 100) < 50) {
        display.fillCircle(20, 16, 4, SH110X_WHITE);
        display.fillCircle(108, 48, 5, SH110X_WHITE);
        display.fillCircle(64, 8, 3, SH110X_WHITE);
      }
      if (elapsed > 900) {
        activeAnim = ANIM_NONE;
        clearLedOverrides();
        digitalWrite(boomPin, LOW);
        state = resumeAfterTransient();
      }
      break;
    }
    default:
      activeAnim = ANIM_NONE;
      clearLedOverrides();
      state = resumeAfterTransient();
      break;
  }

  display.display();
}

bool initOled() {
  // SDA=21 SCL=22 on most ESP32 DevKit boards
  Wire.begin(21, 22);
  delay(250); // battery rails need longer to settle
  Wire.setClock(100000); // slower I2C = happier OLED on weak batteries

  bool ok = display.begin(0x3C, true);
  if (!ok) {
    ok = display.begin(0x3D, true); // some modules use 0x3D
  }
  if (!ok) return false;

  display.setRotation(0);
  display.invertDisplay(false);
  display.setContrast(255);
  display.clearDisplay();
  display.display();
  delay(50);
  display.clearDisplay();
  display.display();
  return true;
}

void hardenBatteryPower() {
  // Stop brownout freezes when USB is yanked or AAA pack sags
  WRITE_PERI_REG(RTC_CNTL_BROWN_OUT_REG, 0);
  setCpuFrequencyMhz(160); // less current than 240 MHz
}

void setup() {
  hardenBatteryPower();
  Serial.begin(115200);
  delay(50);
  // Announce ASAP — Web Serial open resets the board; Maker is already pinging
  Serial.print("JEREMY_OK|");
  Serial.println(FIRMWARE_VERSION);
  pinMode(bootPin, INPUT_PULLUP);
  pinMode(funButton, INPUT_PULLDOWN);
  pinMode(switchPin, INPUT_PULLUP); // open=OFF (HIGH), closed to GND=ON (LOW)
  pinMode(powerPin, OUTPUT);
  pinMode(boomPin, OUTPUT);
  digitalWrite(boomPin, LOW);
  digitalWrite(powerPin, HIGH);
  setupLedPwm();
    setBlue(false);
  setRed(false);
  setYellow(false);
  delay(300); // wait for VIN/battery before touching OLED/WiFi

  testLedsBoot();

  if (!initOled()) {
    // Keep running WiFi even if screen fails — LED blink means OLED init failed
    for (int i = 0; i < 10; i++) {
      setBlue(true); delay(100); setBlue(false); delay(100);
    }
  }

  randomSeed(esp_random());
  buildUniqueSsid();
  clearSparks();
  overlayText[0] = '\0';
  overlayImage[0] = '\0';

  prefs.begin("jeremy", false);
  loadSwitchProgram();
  loadLookProgram();

  // Auth gate rev 1: first boot / this firmware always requires AUTH screen once.
  // Clears stale "paired" from older builds so the OLED is not skipped.
  const int AUTH_GATE_REV = 1;
  int authGate = prefs.getInt("authGate", 0);
  if (authGate < AUTH_GATE_REV) {
    prefs.putBool("paired", false);
    prefs.putBool("needPair", true);
    prefs.putBool("panelOn", false);
    prefs.remove("pairCode");
    prefs.remove("panelCode");
    prefs.putInt("authGate", AUTH_GATE_REV);
  }

  isPaired = prefs.getBool("paired", false);
  needPairingScreen = !isPaired; // first boot / unpaired → always auth

  pairingCode[0] = '\0';
  panelCode[0] = '\0';
  panelCodeActive = false;

  if (!isPaired) {
    String savedPair = prefs.getString("pairCode", "");
    if (savedPair.length() >= 4) {
      strncpy(pairingCode, savedPair.c_str(), sizeof(pairingCode) - 1);
      pairingCode[sizeof(pairingCode) - 1] = '\0';
    } else {
      generateCode(pairingCode, sizeof(pairingCode));
      prefs.putString("pairCode", pairingCode);
    }
  } else {
    // Already authenticated on a previous boot — restore panel session only
    String savedPanel = prefs.getString("panelCode", "");
    if (savedPanel.length() >= 4) {
      strncpy(panelCode, savedPanel.c_str(), sizeof(panelCode) - 1);
      panelCode[sizeof(panelCode) - 1] = '\0';
      panelCodeActive = true;
      prefs.putBool("panelOn", true);
    } else {
      mintNewPanelCode();
    }
  }

  setupWifiAndServer();

  display.clearDisplay();
  display.setTextSize(1);
  display.setTextColor(SH110X_WHITE);
  display.setCursor(10, 28);
  display.print("VERSION: 2.6.52");
  display.display();
  delayServicingUsb(VERSION_DELAY);

  display.clearDisplay();
  display.setCursor(4, 20);
  display.println("PRIVATE TEST UNIT");
  display.setCursor(16, 36);
  display.println("JEREMY CORP LTD");
  display.display();
  delayServicingUsb(2000);

  spamRandomCode(CODE_SPAM_DELAY);

  ox = oy = 0;
  startupTime = millis();
  pickTarget();

  if (!isPaired) {
    state = PAIRING;
    showPairingScreen(); // unskippable auth screen on first boot
  } else {
    state = NORMAL;
  }
}

void loop() {
  dnsServer.processNextRequest();
  server.handleClient();
  serviceUsbProgrammer();

  static bool prevBoot = false;
  bool bootDown = (digitalRead(bootPin) == LOW);
  unsigned long now = millis();
  enforceLedSafety(now);

  // Power switch ON → programmed screen text + selected LEDs until OFF
  if (switchIsOn()) {
    prevBoot = bootDown;
    waitingSecondClick = false;
    forcePop = false;
    activeAnim = ANIM_NONE;
    overlayUntil = 0;
    digitalWrite(boomPin, LOW);
    clearLedOverrides();
    holdSwitchProgramUntilOff();
    return;
  }

  if (state != PLAY_ANIM) {
    if (!ledOverride) setBlue(bootDown);
    else setBlue(ledOverrideOn);
    if (redOverride) setRed(redOverrideOn);
    if (yellowOverride) setYellow(yellowOverrideOn);
  }

  // UNSKIPPABLE AUTH: first boot / unpaired — boot button cannot leave this screen
  if (!isPaired) {
    state = PAIRING;
    prevBoot = bootDown;
    waitingSecondClick = false;
    forcePop = false;
    activeAnim = ANIM_NONE;
    overlayUntil = 0;
    digitalWrite(boomPin, LOW);
    showPairingScreen();
    delay(30);
    return;
  }

  if (bootDown && !prevBoot) {
    if (state == INFO_SCREEN) {
      state = NORMAL;
      waitingSecondClick = false;
    } else if (state == SHOW_OVERLAY || state == PLAY_ANIM) {
      overlayUntil = 0;
      activeAnim = ANIM_NONE;
      clearLedOverrides();
      state = resumeAfterTransient();
      waitingSecondClick = false;
    } else if (waitingSecondClick && (now - lastPressTime < DOUBLE_CLICK_WINDOW)) {
      waitingSecondClick = false;
      openControlPanel();
    } else {
      waitingSecondClick = true;
      lastPressTime = now;
    }
  }
  prevBoot = bootDown;

  if (waitingSecondClick && (now - lastPressTime > DOUBLE_CLICK_WINDOW)) {
    waitingSecondClick = false;
  }

  if (state == SHOW_OVERLAY) {
    if (now >= overlayUntil) {
      state = resumeAfterTransient();
      if (state == SHOW_OVERLAY || state == PLAY_ANIM) state = NORMAL;
    } else {
      renderOverlay();
      delay(30);
      return;
    }
  }

  if (state == PLAY_ANIM) {
    runAnimFrame();
    delay(30);
    return;
  }

  if (now - startupTime < STARTUP_DELAY && state != INFO_SCREEN) {
    drawEyes(size);
    delay(30);
    return;
  }

  if (state == INFO_SCREEN) {
    showInfoScreen();
    delay(30);
    return;
  }

  if (forcePop) {
    forcePop = false;
    size = OVER;
    popTime = now;
    state = EXPLODE;
    setAllLeds(true);
  }

  ox += ((tx - ox) * MOVE_EASE);
  oy += ((ty - oy) * MOVE_EASE);
  driftAngle += 0.02;
  ox += cos(driftAngle) * 0.4;
  oy += sin(driftAngle * 0.8) * 0.4;
  if (!random(0, 45)) pickTarget();

  float half = size * 0.5 + 2;
  ox = __C(ox, -leftX + half, 128 - rightX - half);
  oy = __C(oy, -centerY + half, 64 - centerY - half);

  switch ((int)state) {
    case NORMAL:
      size = __B(size, BASE);
      if (bootDown) { pressStart = now; state = HOLDING; }
      break;
    case HOLDING:
      size = __B(size, GROW);
      if (!bootDown) state = NORMAL;
      if (size > GROW - 1) state = STRETCH;
      break;
    case STRETCH:
      size += ((OVER - size) * 0.3);
      if (size > OVER - 1) { popTime = now; state = EXPLODE; }
      break;
    case EXPLODE:
      display.clearDisplay();
      __F(2) { drawFragments((_ ? rightX : leftX) + ox, centerY + oy, 16); }
      display.display();
      setAllLeds(((now - popTime) / 40) % 2 == 0);
      if (now - popTime > 140) { size = 0; state = REBUILD; clearLedOverrides(); }
      delay(30);
      return;
    case REBUILD:
      size += ((BASE - size) * 0.15);
      if (size > BASE - 1) state = NORMAL;
      break;
    default:
      state = NORMAL;
      break;
  }

  if (digitalRead(funButton) == HIGH) {
    const char* scrollText = "  TEST FUNCTION  ";
    int textWidth = (int)strlen(scrollText) * 6;
    display.setTextSize(1);
    display.setTextColor(SH110X_WHITE);
    for (int x = 128; x > -textWidth; x -= 3) {
      dnsServer.processNextRequest();
      server.handleClient();
      display.clearDisplay();
      display.setCursor(x, 28);
      display.print(scrollText);
      display.display();
      delay(20);
      if (digitalRead(funButton) == LOW) break;
    }
  }

  drawEyes(size);
  delay(30);
}
